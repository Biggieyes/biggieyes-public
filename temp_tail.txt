                            ...(i === arr.length - 1 ? { borderTopRightRadius: 12 } : {}),
                          }}
                        >
                          {h}
                        </th>
                      ))}
                    </tr>
                  </thead>
                  <tbody style={{ color: "#f2f6fb" }}>
                    {hist.length ? (
                      hist.map((r, idx) => (
                        <tr
                          key={`${r.requestId}-${idx}`}
                          style={{
                            background:
                              idx % 2 === 0
                                ? "rgba(255,255,255,.02)"
                                : "rgba(255,255,255,.04)",
                          }}
                        >
                          <td style={{ padding: "10px 12px" }}>{r.time || "—"}</td>
                          <td style={{ padding: "10px 12px", fontFamily: "ui-monospace,Menlo" }}>
                            <Value mono>{short(r.requestId)}</Value>
                          </td>
                          <td style={{ padding: "10px 12px", fontWeight: 800 }}>
                            {String(r.status || "—").toUpperCase()}
                          </td>
                          <td style={{ padding: "10px 12px" }}>{r.confirmations ?? "—"}</td>
                          <td style={{ padding: "10px 12px" }}>{r.words ?? "—"}</td>
                          <td style={{ padding: "10px 12px" }}>
                            {r.tx ? (
                              <GhostBtn color={C.v} onClick={() => onOpenExplorer(r.tx, "tx")}>
                                {short(r.tx)}
                              </GhostBtn>
                            ) : (
                              "—"
                            )}
                          </td>
                        </tr>
                      ))
                    ) : (
                      <tr>
                        <td colSpan={6} style={{ padding: 12, textAlign: "center", opacity: 0.85 }}>
                          —
                        </td>
                      </tr>
                    )}
                  </tbody>
                </table>
              </div>
            </div>
          </Card>
        </div>
      )}

      {/* ===== SETTINGS TAB ===== */}
      {active === "settings" && (
        <div style={{ display: "grid", gap: 16 }}>
          <Card title="?? Request Parameters" icon="??" hue="v" right={<Badge>read-only</Badge>}>
            <KV
              items={[
                { k: "Key Hash", v: params.keyHash || "—", mono: true, tone: "violet", title: params.keyHash },
                { k: "Confirmations", v: params.confirmations ?? "—", mono: true, tone: "violet" },
                { k: "Num Words", v: params.numWords ?? "—", mono: true, tone: "violet" },
                { k: "Callback Gas Limit", v: params.callbackGasLimit ?? "—", mono: true, tone: "violet" },
                ...(params.subscriptionId || params.subId
                  ? [{ k: "Subscription Id", v: params.subscriptionId ?? params.subId, mono: true, tone: "violet" }]
                  : []),
                ...(params.coordinator || params.vrfCoordinator
                  ? [{ k: "Coordinator", v: params.coordinator ?? params.vrfCoordinator, mono: true, tone: "violet" }]
                  : []),
              ]}
            />
          </Card>
        </div>
      )}
      </div>
    </div>
  );
}
